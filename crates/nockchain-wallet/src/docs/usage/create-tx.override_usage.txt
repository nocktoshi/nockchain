nockchain-wallet create-tx [--names <NAMES>] --recipient <RECIPIENT>... [--fee <FEE>] [--refund-pkh <REFUND_PKH>] [--include-data <BOOL>]

# NOTE: --refund-pkh is required when spending from legacy v0 notes. For v1 notes, the refund defaults to the note owner. --include-data defaults to true (pass 'false' to exclude note data).
# NOTE: if --names is omitted, the planner auto-selects spendable v1 notes. If provided, names are treated as a manual selection set.
# NOTE: manual selection may spend either an all-v1 set or an all-v0 set; mixed-version manual sets are rejected.
# NOTE: --fee is optional. If omitted, the planner computes a fee. If provided, it overrides the planner fee (subject to --allow-low-fee).

Examples:
  # Auto-select spendable v1 notes and compute fee
  nockchain-wallet create-tx \
    --recipient '{"kind":"p2pkh","address":"<p2pkh-b58>","amount":10000}'

  # Manually pin notes and optionally override fee
  nockchain-wallet create-tx \
    --names "[first1 last1],[first2 last2]" \
    --recipient '{"kind":"p2pkh","address":"<p2pkh-b58>","amount":10000}' \
    --fee 10

  # --recipient: legacy '<p2pkh>:<amount>' string
  nockchain-wallet create-tx \
    --recipient '<p2pkh-b58>:10000'

  # --recipient: optional `memo` (UTF-8, max ~2048 bytes)
  nockchain-wallet create-tx \
    --recipient '{"kind":"p2pkh","address":"<p2pkh-b58>","amount":10000,"memo":"hello"}'

  # --recipient: optional `blob` (UTF-8; wallet jams a binary data under note-data key `blob`)
  nockchain-wallet create-tx \
    --recipient '{"kind":"p2pkh","address":"<p2pkh-b58>","amount":10000,"blob":"nns/v1/claim/example"}'
